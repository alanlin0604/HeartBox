### 3.1 台灣市場建議：綠界科技 ECPay

#### 為何選擇ECPay？
1. ✅ **台灣市場主流** - 覆蓋率高，用戶熟悉
2. ✅ **支援多種付款方式**:
   - 信用卡（一次付清、分期）
   - ATM轉帳
   - 超商代碼繳費

3. ✅ **定期定額扣款** - 支援訂閱制
4. ✅ **合理費率**:
   - 信用卡: 2.8% - 3.0%

# backend/api/payment.py (新檔案)

from ecpay_payment_sdk.client import EcpayClient
from django.conf import settings
import hashlib
import time

class ECPayService:
    def __init__(self):
        self.merchant_id = settings.ECPAY_MERCHANT_ID
        self.hash_key = settings.ECPAY_HASH_KEY
        self.hash_iv = settings.ECPAY_HASH_IV
        self.test_mode = settings.DEBUG  # 測試/正式環境
        
    def create_subscription_order(self, user, plan):
        """建立訂閱訂單"""
        order_params = {
            'MerchantID': self.merchant_id,
            'MerchantTradeNo': f'SUB{int(time.time())}{user.id}',  # 訂單編號
            'MerchantTradeDate': time.strftime('%Y/%m/%d %H:%M:%S'),
            'PaymentType': 'aio',
            'TotalAmount': int(plan.price),
            'TradeDesc': f'HeartBox {plan.name} 方案訂閱',
            'ItemName': f'{plan.name} 月費',
            'ReturnURL': f'{settings.SITE_URL}/api/payments/ecpay/callback/',
            'ChoosePayment': 'Credit',  # 信用卡
            'EncryptType': 1,
            
            # 定期定額參數
            'PeriodAmount': int(plan.price),
            'PeriodType': 'M',  # M=月, Y=年
            'Frequency': 1,  # 每1個月
            'ExecTimes': 0,  # 0=不限次數
            'PeriodReturnURL': f'{settings.SITE_URL}/api/payments/ecpay/period-callback/',
        }
        
        # 產生檢查碼
        check_mac_value = self.generate_check_mac(order_params)
        order_params['CheckMacValue'] = check_mac_value
        
        return order_params
    
    def generate_check_mac(self, params):
        """產生ECPay檢查碼"""
        sorted_params = sorted(params.items())
        param_str = '&'.join([f'{k}={v}' for k, v in sorted_params])
        param_str = f'HashKey={self.hash_key}&{param_str}&HashIV={self.hash_iv}'
        param_str = urllib.parse.quote_plus(param_str).lower()
        
        return hashlib.sha256(param_str.encode('utf-8')).hexdigest().upper()

class InitiatePaymentView(APIView):
    """發起支付流程"""
    permission_classes = [permissions.IsAuthenticated]
    
    def post(self, request):
        plan_id = request.data.get('plan_id')
        
        try:
            plan = SubscriptionPlan.objects.get(id=plan_id, is_active=True)
        except SubscriptionPlan.DoesNotExist:
            return error_response('plan_not_found', 'Plan not found', 404)
        
        # 免費方案直接訂閱
        if plan.price == 0:
            UserSubscription.objects.update_or_create(
                user=request.user,
                defaults={'plan': plan, 'status': 'active'}
            )
            return Response({'status': 'success', 'payment_required': False})
        
        # 付費方案 - 建立訂單
        ecpay = ECPayService()
        order_params = ecpay.create_subscription_order(request.user, plan)
        
        # 記錄訂單到資料庫
        order = PaymentOrder.objects.create(
            user=request.user,
            plan=plan,
            amount=plan.price,
            currency=plan.currency,
            order_no=order_params['MerchantTradeNo'],
            status='pending',
        )
        
        return Response({
            'status': 'redirect',
            'payment_required': True,
            'payment_url': 'https://payment.ecpay.com.tw/Cashier/AioCheckOut/V5',
            'payment_params': order_params,
        })


class ECPayCallbackView(APIView):
    """ECPay付款回調"""
    permission_classes = [permissions.AllowAny]
    
    def post(self, request):
        # 驗證檢查碼
        ecpay = ECPayService()
        if not ecpay.verify_check_mac(request.data):
            return HttpResponse('0|CheckMacValue Error')
        
        # 處理付款結果
        order_no = request.data.get('MerchantTradeNo')
        rtn_code = request.data.get('RtnCode')
        
        try:
            order = PaymentOrder.objects.get(order_no=order_no)
            
            if rtn_code == '1':  # 付款成功
                order.status = 'paid'
                order.paid_at = timezone.now()
                order.save()
                
                # 啟用訂閱
                UserSubscription.objects.update_or_create(
                    user=order.user,
                    defaults={
                        'plan': order.plan,
                        'status': 'active',
                        'started_at': timezone.now(),
                        'expires_at': timezone.now() + timedelta(days=30),
                    }
                )
                
                # 發送確認信
                send_subscription_confirmation_email(order.user, order.plan)
                
            else:  # 付款失敗
                order.status = 'failed'
                order.save()
            
            return HttpResponse('1|OK')
            
        except PaymentOrder.DoesNotExist:
            return HttpResponse('0|Order Not Found')

export const initiatePayment = (planId) => {

// frontend/src/pages/PricingPage.jsx (修改)

const handleSubscribe = async (planId) => {
  try {
    const { data } = await initiatePayment(planId)
    
    if (!data.payment_required) {
      // 免費方案直接完成
      toast.success(t('subscription.subscribed'))
      // 重新載入訂閱狀態
      return
    }
    
    // 付費方案 - 導向ECPay
    if (data.payment_url && data.payment_params) {
      // 建立表單並提交到ECPay
      const form = document.createElement('form')
      form.method = 'POST'
      form.action = data.payment_url
      
      Object.entries(data.payment_params).forEach(([key, value]) => {
        const input = document.createElement('input')
        input.type = 'hidden'
        input.name = key
        input.value = value
        form.appendChild(input)
      })
      
      document.body.appendChild(form)
      form.submit()
    }
  } catch (error) {
    toast.error(t('subscription.subscribeFailed'))
  }
}

# backend/api/models.py (新增)

class PaymentOrder(models.Model):
    STATUS_CHOICES = [
        ('pending', 'Pending'),
        ('paid', 'Paid'),
        ('failed', 'Failed'),
        ('refunded', 'Refunded'),
        ('cancelled', 'Cancelled'),
    ]
    
    user = models.ForeignKey(settings.AUTH_USER_MODEL, on_delete=models.CASCADE)
    plan = models.ForeignKey(SubscriptionPlan, on_delete=models.PROTECT)
    order_no = models.CharField(max_length=100, unique=True)
    amount = models.DecimalField(max_digits=10, decimal_places=2)
    currency = models.CharField(max_length=3, default='TWD')
    status = models.CharField(max_length=20, choices=STATUS_CHOICES)
    
    # ECPay相關
    payment_type = models.CharField(max_length=50, default='Credit')  # Credit/ATM/CVS
    trade_no = models.CharField(max_length=100, blank=True)  # ECPay交易編號
    period_no = models.CharField(max_length=100, blank=True)  # 定期定額編號
    
    created_at = models.DateTimeField(auto_now_add=True)
    paid_at = models.DateTimeField(null=True, blank=True)
    
    class Meta:
        ordering = ['-created_at']
    
    def __str__(self):
        return f'{self.order_no} - {self.user.username} - {self.status}'


class PaymentTransaction(models.Model):
    """記錄每次扣款"""
    order = models.ForeignKey(PaymentOrder, on_delete=models.CASCADE, related_name='transactions')
    transaction_no = models.CharField(max_length=100)
    amount = models.DecimalField(max_digits=10, decimal_places=2)
    status = models.CharField(max_length=20)
    period_date = models.DateField()  # 扣款日期
    created_at = models.DateTimeField(auto_now_add=True)
    
    class Meta:
        ordering = ['-created_at']

- **解決**: 
  - 寬限期機制（3-7天）
  - Email通知用戶更新付款方式
  - 自動重試（1-2次）

#### Webhook延遲/失敗
- 網路問題
- 服務器當機
- **解決**:
  - 實作冪等性（重複請求不影響）
  - 訂單狀態對賬機制
  - 手動處理介面

#### 安全性
- 中間人攻擊
- 重放攻擊
- **解決**:
  - HTTPS強制
  - CheckMacValue驗證
  - 時間戳檢查

### 6.3 用戶體驗風險

#### 付款流程中斷
- 用戶關閉瀏覽器
- 網路斷線
- **解決**:
  - 訂單查詢功能
  - 重新進入付款流程
  - 客服協助處理

#### 退款糾紛
- 用戶不滿意服務
- 重複扣款
- **解決**:
  - 明確的退款政策
  - 7天鑑賞期（視法規）
  - 客服處理流程
